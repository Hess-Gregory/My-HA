    28	# climate.couloir / climate.living / climate.salle_de_bain - voir section 9).
    29	#
    30	# LOGIQUE GÉNÉRALE :
    31	#   - Chaque pièce a SON PROPRE mode (contrairement à la 1ère version qui
    32	#     avait un mode unique pour tout l'appart) car vos horaires réels font
    33	#     que les pièces ne sont pas toutes dans le même état en même temps
    34	#     (ex. 16h40 : salon/SdB/chambre enfant repassent en Confort, mais la
    35	#     chambre adulte reste en Éco jusqu'à 22h30 puisqu'elle n'est pas
    36	#     utilisée avant le coucher).
    37	#   - input_boolean.routine_activee est l'interrupteur maître : coupe-le
    38	#     pendant les vacances ou en cas de maladie pour désactiver tout le
    39	#     planning automatique (l'appart bascule alors en Hors-gel). Rallume-le
    40	#     à ton retour pour reprendre la routine normale.
    41	#   - Le couloir n'a volontairement PAS de planning horaire : il reste
    42	#     toujours sur "Éco" (jamais chaud, jamais gelé), sauf en Hors-gel
    43	#     pendant les vacances.
    44	#   - Chambre enfants, chambre adulte et salle de bain ne sont PLUS chauffées
    45	#     automatiquement "en journée" (quand elles sont inoccupées) : chambre
    46	#     enfants et chambre adulte restent en Éco/Nuit sauf sieste déclenchée
    47	#     manuellement (boutons section 9), et la salle de bain reste en Éco en
    48	#     permanence sauf activation manuelle juste avant de s'en servir (boutons
    49	#     section 9 aussi). Dans tous les cas, la prochaine transition programmée
    50	#     (retour du soir, coucher...) reprend la main normalement, donc pas
    51	#     besoin de désactiver quoi que ce soit après une sieste ou une douche :
    52	#     au pire ça reste chaud jusqu'au prochain horaire programmé.
    53	#   - La chambre adulte est pré-chauffée à 22h30 TOUS LES JOURS (pas
    54	#     seulement en semaine) pour ne jamais se coucher dans des draps froids.

   754	# Interpolation linéaire entre deux points de référence réglables via les
   755	# input_number "chauffage_courbe_*" (section 1bis) : plus il fait froid

   758	# la chaudière plus chaud que nécessaire (meilleur rendement en mode
   759	# condensation = moins de gaz), sans jamais sous-chauffer par grand froid.
   760	# Se contente de RENVOYER LA VALEUR ACTUELLE de number.boiler_selflowtemp si
   761	# weather.forecast_maison est indisponible, pour ne jamais forcer un
   762	# changement basé sur une donnée manquante.

   768	        state: >

   770	          {% if t_ext is none %}
   771	            {{ states('number.boiler_selflowtemp') }}
   772	          {% else %}
   773	            {% set t_ext = t_ext | float %}
   774	            {% set x1 = states('input_number.chauffage_courbe_temp_ext_froid') | float(-7) %}
   775	            {% set y1 = states('input_number.chauffage_courbe_depart_froid') | float(55) %}
   776	            {% set x2 = states('input_number.chauffage_courbe_temp_ext_doux') | float(15) %}
   777	            {% set y2 = states('input_number.chauffage_courbe_depart_doux') | float(32) %}
   778	            {% if t_ext <= x1 %}
   779	              {{ y1 }}
   780	            {% elif t_ext >= x2 %}
   781	              {{ y2 }}
   782	            {% else %}
   783	              {% set pente = (y2 - y1) / (x2 - x1) %}
   784	              {{ (y1 + pente * (t_ext - x1)) | round(1) }}

  1385	# importe le mode. Hystérésis (0.3°C sous consigne pour rallumer, 0.2°C
